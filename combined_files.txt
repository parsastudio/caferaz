"use client";

import React, { createContext, useContext, useState } from "react";

export interface CheckoutData {
  productId: string;
  selectedWeight: number;
  selectedBlend: string;
  selectedGrind: string;
  quantity: number;
  finalPrice: number;
}

interface CheckoutContextType {
  isOpen: boolean;
  checkoutData: CheckoutData | null;
  openCheckout: (data: CheckoutData) => void;
  closeCheckout: () => void;
}

const CheckoutContext = createContext<CheckoutContextType | undefined>(
  undefined,
);

export function CheckoutProvider({ children }: { children: React.ReactNode }) {
  const [isOpen, setIsOpen] = useState(false);
  const [checkoutData, setCheckoutData] = useState<CheckoutData | null>(null);

  const openCheckout = (data: CheckoutData) => {
    setCheckoutData(data);
    setIsOpen(true);
  };

  const closeCheckout = () => {
    setIsOpen(false);
  };

  return (
    <CheckoutContext.Provider
      value={{ isOpen, checkoutData, openCheckout, closeCheckout }}
    >
      {children}
    </CheckoutContext.Provider>
  );
}

export function useCheckout() {
  const context = useContext(CheckoutContext);
  if (!context) {
    throw new Error("useCheckout must be used within a CheckoutProvider");
  }
  return context;
}

"use client";

import React, { useState, useOptimistic, useTransition } from "react";
import { MessageSquare, Star, Plus, Clock } from "lucide-react";
import { Tabs, TabsList, TabsTrigger, TabsContent } from "./ui/tabs";
import { Button } from "./ui/button";
import CommentDialog from "./CommentDialog";

export interface UIComment {
  id: string;
  first_name: string;
  last_name: string;
  rating: number | null;
  comment_text: string;
  created_at: string;
  is_pending?: boolean;
  replies?: UIComment[];
  parent_id_temp?: string | null;
}

interface ProductTabsSectionProps {
  productId: string;
  description: string;
  initialComments: UIComment[];
}

export default function ProductTabsSection({
  productId,
  description,
  initialComments,
}: ProductTabsSectionProps) {
  const [isCommentDialogOpen, setIsCommentDialogOpen] = useState(false);
  const [replyParentId, setReplyParentId] = useState<string | null>(null);
  const [isPending, startTransition] = useTransition();

  const [optimisticComments, addOptimisticComment] = useOptimistic(
    initialComments,
    (state, newComment: UIComment) => {
      if (newComment.parent_id_temp) {
        return state.map((c) => {
          if (c.id === newComment.parent_id_temp) {
            return {
              ...c,
              replies: [...(c.replies || []), newComment],
            };
          }
          return c;
        });
      }
      return [newComment, ...state];
    },
  );

  const handleCommentSubmitted = (newComment: {
    first_name: string;
    comment_text: string;
    rating: number | null;
    parentId: string | null;
  }) => {
    const newLocalComment: UIComment = {
      id: `local_new_${Date.now()}`,
      first_name: newComment.first_name,
      last_name: "",
      rating: newComment.rating,
      comment_text: newComment.comment_text,
      created_at: new Date().toISOString(),
      is_pending: true,
      replies: [],
      parent_id_temp: newComment.parentId,
    };

    addOptimisticComment(newLocalComment);
  };

  const openNewCommentDialog = () => {
    setReplyParentId(null);
    setIsCommentDialogOpen(true);
  };

  const openReplyDialog = (parentId: string) => {
    setReplyParentId(parentId);
    setIsCommentDialogOpen(true);
  };

  return (
    <div
      className="mt-16 border-t border-stone-800/80 pt-10 text-right"
      dir="rtl"
    >
      <Tabs defaultValue="comments" className="flex flex-col w-full" dir="rtl">
        <TabsList
          className="bg-stone-900 border border-stone-800 p-1 mb-8 flex w-full max-w-md"
          dir="rtl"
        >
          <TabsTrigger
            value="comments"
            className="flex-1 flex items-center justify-center gap-2 cursor-pointer py-2.5"
          >
            <MessageSquare className="w-4 h-4" />
            نظرات مشتریان ({optimisticComments?.length || 0})
          </TabsTrigger>
          <TabsTrigger
            value="description"
            className="flex-1 flex items-center justify-center gap-2 cursor-pointer py-2.5"
          >
            توضیحات تخصصی محصول
          </TabsTrigger>
        </TabsList>

        <TabsContent
          value="comments"
          className="space-y-6 w-full text-right"
          dir="rtl"
        >
          <div
            className="flex flex-row items-center justify-between w-full"
            dir="rtl"
          >
            <h3 className="text-lg font-black text-stone-200">
              نظرات و پرسش‌های کاربران
            </h3>
            <Button
              onClick={openNewCommentDialog}
              className="bg-gold-500 hover:bg-gold-600 text-stone-950 font-bold cursor-pointer"
            >
              <Plus className="w-4 h-4 ml-1.5" />
              ثبت دیدگاه جدید
            </Button>
          </div>

          {optimisticComments && optimisticComments.length > 0 ? (
            <div
              className="divide-y divide-stone-800/80 bg-stone-900/20 p-6 rounded-2xl border border-stone-900/60 text-right"
              dir="rtl"
            >
              {optimisticComments.map((comment: UIComment) => (
                <CommentNode
                  key={comment.id}
                  comment={comment}
                  onReply={openReplyDialog}
                />
              ))}
            </div>
          ) : (
            <div className="text-center py-12 bg-stone-900/20 rounded-2xl border border-dashed border-stone-800">
              <p className="text-sm text-stone-500">
                تاکنون دیدگاهی ثبت نشده است. اولین دیدگاه را شما ثبت کنید!
              </p>
            </div>
          )}
        </TabsContent>

        <TabsContent
          value="description"
          className="bg-stone-900/20 p-6 rounded-2xl border border-stone-900/60 text-stone-300 leading-relaxed text-sm w-full text-right"
          dir="rtl"
        >
          <p>{description || "توضیحات تکمیلی برای این محصول ثبت نشده است."}</p>
        </TabsContent>
      </Tabs>

      <CommentDialog
        isOpen={isCommentDialogOpen}
        onOpenChange={setIsCommentDialogOpen}
        productId={productId}
        replyParentId={replyParentId}
        onCommentSubmitted={handleCommentSubmitted}
        startTransition={startTransition}
      />
    </div>
  );
}

interface CommentNodeProps {
  comment: UIComment;
  onReply: (parentId: string) => void;
}

function CommentNode({ comment, onReply }: CommentNodeProps) {
  return (
    <div className="py-5 space-y-3 text-right" dir="rtl">
      <div
        className="flex flex-row items-center justify-between w-full"
        dir="rtl"
      >
        <div className="flex items-center gap-2.5" dir="rtl">
          <div className="w-9 h-9 rounded-full bg-stone-800 border border-stone-700/60 flex items-center justify-center text-xs font-black text-gold-500">
            {comment.first_name?.[0] || "ک"}
          </div>
          <div className="flex flex-col">
            <span className="font-bold text-stone-200 text-sm flex items-center gap-2">
              {comment.first_name} {comment.last_name}
              {comment.is_pending && (
                <span className="text-[10px] bg-gold-500/10 border border-gold-500/20 text-gold-500 px-2 py-0.5 rounded-full flex items-center gap-1">
                  <Clock className="w-3 h-3" /> در انتظار تایید
                </span>
              )}
            </span>
          </div>
          {comment.rating && (
            <div className="flex items-center gap-1 bg-gold-500/10 border border-gold-500/25 text-gold-500 px-2 py-0.5 rounded-lg text-xs font-bold">
              <Star className="w-3.5 h-3.5 fill-gold-500 text-gold-500" />
              <span>{comment.rating}</span>
            </div>
          )}
        </div>
        <span className="text-xs text-stone-500">
          {new Date(comment.created_at).toLocaleDateString("fa-IR")}
        </span>
      </div>

      <p
        className="text-sm text-stone-300 pr-11 leading-relaxed text-right"
        dir="rtl"
      >
        {comment.comment_text}
      </p>

      <div className="pr-11">
        <button
          type="button"
          onClick={() => onReply(comment.id)}
          className="text-xs text-gold-500 hover:text-gold-400 hover:underline transition-colors font-bold cursor-pointer"
        >
          پاسخ به این نظر
        </button>
      </div>

      {comment.replies && comment.replies.length > 0 && (
        <div className="mr-9 pr-4 border-r border-stone-800/80 space-y-3 mt-4">
          {comment.replies.map((reply: UIComment) => (
            <CommentNode key={reply.id} comment={reply} onReply={onReply} />
          ))}
        </div>
      )}
    </div>
  );
}

"use client";

import React, { useState } from "react";
import { updateProductDetailsAction } from "@/app/actions/product";
import { ProductWithPricing } from "@/lib/mock-db";
import { toast } from "sonner";
import { Loader2, Sparkles } from "lucide-react";

interface AdminPricingFormProps {
  initialProduct: ProductWithPricing;
}

export default function AdminPricingForm({
  initialProduct,
}: AdminPricingFormProps) {
  const [name, setName] = useState(initialProduct.name || "");
  const [description, setDescription] = useState(
    initialProduct.description || "",
  );

  const [arabicaRaw, setArabicaRaw] = useState<number>(
    initialProduct.pricing?.pricePerKgArabica || 2200000,
  );
  const [robustaRaw, setRobustaRaw] = useState<number>(
    initialProduct.pricing?.pricePerKgRobusta || 1800000,
  );

  const [stockKg, setStockKg] = useState<number>(
    (initialProduct.stock_grams || 150000) / 1000,
  );

  const [isSubmitting, setIsSubmitting] = useState(false);

  const formatNumberWithCommas = (value: number) => {
    if (value === 0) return "";
    return value.toLocaleString("en-US");
  };

  const handleNumberChange = (
    rawValue: string,
    setter: React.Dispatch<React.SetStateAction<number>>,
  ) => {
    const cleanValue = rawValue.replace(/[^0-9]/g, "");
    const numericValue = cleanValue === "" ? 0 : parseInt(cleanValue, 10);
    setter(numericValue);
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (
      !name.trim() ||
      !description.trim() ||
      arabicaRaw <= 0 ||
      robustaRaw <= 0 ||
      stockKg < 0
    ) {
      toast.error("لطفاً تمامی فیلدها را با مقادیر معتبر تکمیل کنید.");
      return;
    }

    setIsSubmitting(true);
    try {
      const result = await updateProductDetailsAction({
        productId: initialProduct.id,
        name,
        description,
        pricePerKgArabica: arabicaRaw,
        pricePerKgRobusta: robustaRaw,
        stockGrams: stockKg * 1000,
      });

      if (result.success) {
        toast.success(
          "تنظیمات، نرخ‌ها و موجودی انبار با موفقیت به‌روزرسانی شدند.",
        );
      } else {
        toast.error(result.error || "خطا در ثبت تغییرات.");
      }
    } catch {
      toast.error("خطای سیستمی رخ داد.");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <div className="bg-stone-900/30 border border-stone-800 p-6 rounded-2xl space-y-6">
      <form onSubmit={handleSubmit} className="space-y-5">
        <div className="space-y-2">
          <label className="text-xs font-bold text-stone-300 block">
            نام تجاری و عنوان کالا:
          </label>
          <input
            type="text"
            value={name}
            onChange={(e) => setName(e.target.value)}
            className="w-full bg-stone-950 border border-stone-800 focus:border-gold-500 focus:ring-1 rounded-xl px-4 py-3 text-sm text-stone-100 outline-none transition-all font-bold"
          />
        </div>

        <div className="space-y-2">
          <label className="text-xs font-bold text-stone-300 block">
            توضیحات تخصصی و معرفی محصول:
          </label>
          <textarea
            rows={4}
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            className="w-full bg-stone-950 border border-stone-800 focus:border-gold-500 focus:ring-1 rounded-xl px-4 py-3 text-sm text-stone-100 outline-none transition-all resize-none leading-relaxed"
          />
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
          <div className="space-y-2">
            <label className="text-xs font-bold text-stone-300 block">
              قیمت هر کیلوگرم عربیکا (تومان):
            </label>
            <input
              type="text"
              inputMode="numeric"
              value={formatNumberWithCommas(arabicaRaw)}
              onChange={(e) =>
                handleNumberChange(e.target.value, setArabicaRaw)
              }
              className="w-full bg-stone-950 border border-stone-800 focus:border-gold-500 focus:ring-1 rounded-xl px-4 py-3 text-sm text-gold-400 outline-none transition-all font-bold text-left"
              dir="ltr"
            />
            {arabicaRaw > 0 && (
              <span className="text-[10px] text-stone-500 font-bold flex items-center gap-1 mt-1 pr-1">
                <Sparkles className="w-3 h-3 text-gold-500" />
                معادل:{" "}
                <span className="text-stone-300">
                  {arabicaRaw.toLocaleString("fa-IR")} تومان
                </span>
              </span>
            )}
          </div>

          <div className="space-y-2">
            <label className="text-xs font-bold text-stone-300 block">
              قیمت هر کیلوگرم روبوستا (تومان):
            </label>
            <input
              type="text"
              inputMode="numeric"
              value={formatNumberWithCommas(robustaRaw)}
              onChange={(e) =>
                handleNumberChange(e.target.value, setRobustaRaw)
              }
              className="w-full bg-stone-950 border border-stone-800 focus:border-gold-500 focus:ring-1 rounded-xl px-4 py-3 text-sm text-gold-400 outline-none transition-all font-bold text-left"
              dir="ltr"

                معادل:{" "}

"use client";

import React, { useEffect, useState, useRef } from "react";
import { Smartphone, Loader2 } from "lucide-react";
import { useFormContext } from "react-hook-form";
import { Button } from "../ui/button";
import { CheckoutFormValues } from "./schema";

interface OtpStepProps {
  phone: string;
  onBack: () => void;
  onSubmit: (e: React.FormEvent) => void;
  isSubmitting: boolean;
  finalPrice: number;
}

export default function OtpStep({
  phone,
  onBack,
  onSubmit,
  isSubmitting,
  finalPrice,
}: OtpStepProps) {
  const {
    setValue,
    formState: { errors },
  } = useFormContext<CheckoutFormValues>();

  const [otpDigits, setOtpDigits] = useState<string[]>(["", "", "", ""]);
  const otpInputsRef = useRef<(HTMLInputElement | null)[]>([]);

  const [timer, setTimer] = useState(120);
  const [isResendDisabled, setIsResendDisabled] = useState(true);

  useEffect(() => {
    let interval: NodeJS.Timeout;
    if (timer > 0) {
      interval = setInterval(() => {
        setTimer((prev) => prev - 1);
      }, 1000);
    } else {
      setIsResendDisabled(false);
    }
    return () => clearInterval(interval);
  }, [timer]);

  const handleOtpChange = (value: string, index: number) => {
    const cleanValue = value.replace(/[^0-9]/g, "");

    const newOtp = [...otpDigits];
    newOtp[index] = cleanValue
      ? cleanValue.substring(cleanValue.length - 1)
      : "";
    setOtpDigits(newOtp);

    const fullOtpString = newOtp.join("");
    setValue("otp", fullOtpString, { shouldValidate: true });

    if (cleanValue && index < 3) {
      otpInputsRef.current[index + 1]?.focus();
    }
  };

  const handleOtpKeyDown = (
    e: React.KeyboardEvent<HTMLInputElement>,
    index: number,
  ) => {
    if (e.key === "Backspace") {
      const newOtp = [...otpDigits];
      if (otpDigits[index] === "") {
        if (index > 0) {
          newOtp[index - 1] = "";
          setOtpDigits(newOtp);
          setValue("otp", newOtp.join(""), { shouldValidate: true });
          otpInputsRef.current[index - 1]?.focus();
        }
      } else {
        newOtp[index] = "";
        setOtpDigits(newOtp);
        setValue("otp", newOtp.join(""), { shouldValidate: true });
      }
    }
  };

  const handleOtpPaste = (e: React.ClipboardEvent<HTMLInputElement>) => {
    e.preventDefault();
    const pastedData = e.clipboardData
      .getData("text")
      .replace(/[^0-9]/g, "")
      .substring(0, 4);
    if (pastedData.length === 4) {
      const digits = pastedData.split("");
      setOtpDigits(digits);
      setValue("otp", pastedData, { shouldValidate: true });
      otpInputsRef.current[3]?.focus();
    }
  };

  const handleResendCode = () => {
    const cleared = ["", "", "", ""];
    setOtpDigits(cleared);
    setValue("otp", "", { shouldValidate: true });
    setTimer(120);
    setIsResendDisabled(true);
    otpInputsRef.current[0]?.focus();
  };

  const formatTime = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins}:${secs < 10 ? "0" : ""}${secs}`;
  };

  return (
    <form onSubmit={onSubmit} className="space-y-5 text-right" dir="rtl">
      <div className="text-center space-y-2">
        <div className="w-12 h-12 rounded-full bg-gold-500/10 border border-gold-500/20 text-gold-500 flex items-center justify-center mx-auto mb-1">
          <Smartphone className="w-6 h-6" />
        </div>
        <p className="text-sm font-black text-stone-200">
          کد تایید را وارد کنید
        </p>
        <p className="text-[11px] text-stone-400 leading-relaxed max-w-xs mx-auto">
          یک پیامک حاوی کد تایید برای شماره{" "}
          <span className="text-gold-400 font-bold tracking-wider" dir="ltr">
            {phone}
          </span>{" "}
          ارسال گردید.
        </p>
      </div>

      <div className="flex justify-center gap-3" dir="ltr">
        {otpDigits.map((digit, index) => (
          <input
            key={index}
            type="text"
            inputMode="numeric"
            pattern="[0-9]*"
            maxLength={1}
            value={digit}
            ref={(el) => {
              otpInputsRef.current[index] = el;
            }}
            onChange={(e) => handleOtpChange(e.target.value, index)}
            onKeyDown={(e) => handleOtpKeyDown(e, index)}
            onPaste={handleOtpPaste}
            className={`w-12 h-14 text-center text-xl font-black bg-stone-950 border-2 rounded-xl outline-none text-gold-400 transition-all shadow-inner ${
              errors.otp
                ? "border-destructive/60 focus:border-destructive focus:ring-destructive/30"
                : "border-stone-800/80 focus:border-gold-500 focus:ring-gold-500"
            }`}
          />
        ))}
      </div>

      {errors.otp && (
        <p className="text-[10px] text-destructive font-bold text-center mt-1">
          {errors.otp.message}
        </p>
      )}

      <div className="text-center">
        {timer > 0 ? (
          <span className="text-xs text-stone-500 font-medium">
            امکان ارسال مجدد کد در:{" "}
            <span className="text-stone-300 font-bold">
              {formatTime(timer)}
            </span>
          </span>
        ) : (
          <button
            type="button"
            disabled={isResendDisabled}
            onClick={handleResendCode}
            className="text-xs text-gold-500 hover:text-gold-400 font-bold hover:underline cursor-pointer"
          >
            ارسال مجدد کد تایید
          </button>
        )}
      </div>

      <div className="border-t border-stone-850 pt-4 mt-6 flex items-center justify-between">
        <div className="text-right">
          <span className="text-[10px] text-stone-500 block font-bold">
            پرداخت در محل:
          </span>
          <span className="text-sm font-black text-gold-500">
            {finalPrice.toLocaleString()} تومان
          </span>
        </div>
        <div className="flex gap-2">
          <Button
            type="button"
            variant="ghost"
            onClick={onBack}
            className="text-stone-400 hover:text-stone-200 cursor-pointer text-xs"
            disabled={isSubmitting}
          >
            ویرایش آدرس
          </Button>
          <Button
            type="submit"
            disabled={isSubmitting}
            className="bg-gold-500 hover:bg-gold-600 text-stone-950 font-bold px-5 py-2.5 rounded-xl flex items-center gap-1.5 cursor-pointer"
          >
            {isSubmitting ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                درحال بررسی...
              </>
            ) : (
              "تایید نهایی و ثبت"
            )}
          </Button>
        </div>
      </div>
    </form>
  );
}

export function getCommentsByProductId(

): Comment[] {
  const list = MOCK_COMMENTS.filter((c) => c.productId === productId);

  const filterApproved = (commentsList: Comment[]): Comment[] => {
    return commentsList
      .filter((c) => includeUnapproved || c.status === "approved" || !c.status)

export function addComment(
  productId: string,
  data: {
    first_name: string;
    last_name: string;
    rating: number | null;
    comment_text: string;
    parentId: string | null;
    status?: "pending" | "approved" | "rejected";
  },
): Comment {
  const newId = `comm_${Date.now()}`;
  const createdAt = new Date().toISOString();
  const defaultStatus = data.status || "pending";

  if (data.parentId) {
    const reply: Comment = {
      id: newId,
      first_name: data.first_name,
      last_name: data.last_name,
      rating: null,
      comment_text: data.comment_text,
      created_at: createdAt,
      replies: [],
      status: defaultStatus,
    };

    const findAndPush = (list: Comment[]): boolean => {
      for (const c of list) {
        if (c.id === data.parentId) {
          if (!c.replies) c.replies = [];
          c.replies.push(reply);
          return true;
        }
        if (c.replies && findAndPush(c.replies)) {
          return true;
        }
      }
      return false;
    };

    findAndPush(MOCK_COMMENTS);
    return reply;
  } else {
    const comment: Comment = {
      id: newId,
      productId,
      first_name: data.first_name,
      last_name: data.last_name,
      rating: data.rating,
      comment_text: data.comment_text,
      created_at: createdAt,
      replies: [],
      status: defaultStatus,
    };
    MOCK_COMMENTS.unshift(comment);
    return comment;
  }
}

export function deleteComment(commentId: string): boolean {
  const index = MOCK_COMMENTS.findIndex((c) => c.id === commentId);
  if (index !== -1) {
    MOCK_COMMENTS.splice(index, 1);
    return true;
  }
  const findAndRemove = (list: Comment[]): boolean => {
    for (let i = 0; i < list.length; i++) {
      if (list[i].id === commentId) {
        list.splice(i, 1);
        return true;
      }
      if (list[i].replies && findAndRemove(list[i].replies)) {
        return true;
      }
    }
    return false;
  };
  return findAndRemove(MOCK_COMMENTS);
}

export function calculatePriceMetrics(
  pricing: { pricePerKgArabica: number; pricePerKgRobusta: number } | null,
  blendProfile: BlendProfile,
  weight: number,
  quantity: number,
): PriceMetrics {
  const activePricing = pricing || {
    pricePerKgArabica: 2200000,
    pricePerKgRobusta: 1800000,
  };

  const arabicaRatio = blendProfile.arabica / 100;
  const robustaRatio = blendProfile.robusta / 100;

  const pricePerKg =
    activePricing.pricePerKgArabica * arabicaRatio +
    activePricing.pricePerKgRobusta * robustaRatio;

  const pricePerGram = pricePerKg / 1000;
  const finalPrice = Math.round(pricePerGram * weight * quantity);

  return { originalPrice: finalPrice, finalPrice };
}

export function calculateRemainingStock(
  stockGrams: number | undefined,
  weight: number,
): number {
  const stock = stockGrams !== undefined ? stockGrams : 150000;
  return Math.floor(stock / weight);
}

import { useRouter } from "next/navigation";
import {
  sendUserOtpAction,
  loginUserWithOtpAction,
  checkPhoneRegisteredAction,
  registerUserAction,
} from "@/app/actions/auth";
import {
  Smartphone,
  ArrowRight,
  Loader2,
  MessageSquare,
  User,
  MapPin,
} from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";
import { Button } from "@/components/ui/button";

function LoginForm() {
  const router = useRouter();
  const [mode, setMode] = useState<"login" | "register">("login");
  const [step, setStep] = useState<"phone_or_details" | "otp">(
    "phone_or_details",
  );

  const [phone, setPhone] = useState("");
  const [name, setName] = useState("");
  const [address, setAddress] = useState("");
  const [otp, setOtp] = useState("");

  const [error, setError] = useState<string | null>(null);
  const [isPending, startTransition] = useTransition();

  const handlePhoneSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);

    if (!/^09[0-9]{9}$/.test(phone)) {
      setError("لطفاً یک شماره موبایل معتبر (مانند ۰۹۱۲۳۴۵۶۷۸۹) وارد کنید.");
      return;
    }

    if (mode === "register" && (!name.trim() || !address.trim())) {
      setError("لطفاً تمامی فیلدهای ثبت‌نام را به درستی پر کنید.");
      return;
    }

    startTransition(async () => {
      const checkRes = await checkPhoneRegisteredAction({ phone });
      if (checkRes.success) {
        if (mode === "login" && !checkRes.registered) {
          setMode("register");
          setError(
            "شماره همراه شما ثبت نشده است. لطفاً فرم ثبت‌نام زیر را تکمیل فرمایید.",
          );
          return;
        }
        if (mode === "register" && checkRes.registered) {
          setMode("login");
          setError("این شماره از قبل ثبت شده است. به بخش ورود منتقل شدید.");
          return;
        }

        const otpRes = await sendUserOtpAction({ phone });
        if (otpRes.success) {
          setStep("otp");
        } else {
          setError(otpRes.error || "خطا در ارسال کد تایید.");
        }
      }
    });
  };

  const handleOtpVerify = (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);

    startTransition(async () => {
      try {
        if (mode === "register") {
          const regRes = await registerUserAction({
            phone,
            name,
            address,
            otp,
          });
          if (regRes.success) {
            window.location.href = "/profile";
          } else {
            setError(regRes.error || "خطایی در ثبت‌نام رخ داد.");
          }
        } else {
          const loginRes = await loginUserWithOtpAction({ phone, otp });
          if (loginRes.success) {
            window.location.href = "/profile";
          } else {
            setError(loginRes.error || "کد تایید اشتباه است.");
          }
        }
      } catch {
        setError("خطای سیستمی رخ داد.");
      }
    });
  };

  return (
    <AnimatePresence mode="wait">
      {step === "phone_or_details" ? (
        <motion.form
          key={mode}
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          exit={{ opacity: 0, y: -10 }}
          onSubmit={handlePhoneSubmit}
          className="space-y-4"
        >
          <div className="flex bg-stone-950 p-1 rounded-xl border border-stone-850">
            <button
              type="button"
              onClick={() => {
                setError(null);
                setMode("login");
              }}
              className={`flex-1 py-2 text-xs font-bold rounded-lg transition-all cursor-pointer ${
                mode === "login"
                  ? "bg-gold-500 text-stone-950"
                  : "text-stone-400 hover:text-stone-200"
              }`}
            >
              ورود سریع
            </button>
            <button
              type="button"
              onClick={() => {
                setError(null);
                setMode("register");
              }}
              className={`flex-1 py-2 text-xs font-bold rounded-lg transition-all cursor-pointer ${
                mode === "register"

            >

          </div>

          <div className="space-y-1.5 text-right">

              <Smartphone className="w-3.5 h-3.5 text-stone-500" />
              شماره تلفن همراه
            </label>
            <input
              type="tel"
              value={phone}
              onChange={(e) => setPhone(e.target.value)}
              placeholder="09123456789"
              className="w-full bg-stone-950 border border-stone-800 focus:border-gold-500 focus:ring-1 focus:ring-gold-500 rounded-xl px-4 py-3 text-sm text-stone-100 outline-none text-left font-sans"
              dir="ltr"

            />

                  <User className="w-3.5 h-3.5 text-stone-500" />
                  نام و نام خانوادگی
                </label>
                <input

"use client";

import React, { useState, useTransition } from "react";
import { updateAdminPasswordAction } from "@/app/actions/auth";
import { toast } from "sonner";
import { Loader2, Check } from "lucide-react";
import { Button } from "@/components/ui/button";

export default function ChangePasswordPage() {
  const [oldPass, setOldPass] = useState("");
  const [newPass, setNewPass] = useState("");
  const [confirmPass, setConfirmPass] = useState("");
  const [isPending, startTransition] = useTransition();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    if (!oldPass || !newPass || !confirmPass) {
      toast.error("لطفاً تمامی فیلدها را تکمیل کنید.");
      return;
    }

    if (newPass !== confirmPass) {
      toast.error("رمز عبور جدید با تاییدیه آن مطابقت ندارد.");
      return;
    }

    startTransition(async () => {
      try {
        const result = await updateAdminPasswordAction({ oldPass, newPass });
        if (result.success) {
          toast.success(result.message || "تغییر رمز عبور با موفقیت انجام شد.");
          setOldPass("");
          setNewPass("");
          setConfirmPass("");
        } else {
          toast.error(result.error || "خطایی رخ داد.");
        }
      } catch {
        toast.error("خطا در ارتباط با سرور.");
      }
    });
  };

  return (
    <div className="space-y-8 text-right max-w-xl" dir="rtl">
      <div>
        <h1 className="text-2xl font-black text-stone-100">
          تغییر رمز عبور مدیریت
        </h1>
        <p className="text-xs text-stone-400 mt-1">
          به‌روزرسانی کلید امنیتی دسترسی به پنل مدیریت کافه راز
        </p>
      </div>

      <div className="bg-stone-900/30 border border-stone-800 p-6 rounded-2xl">
        <form onSubmit={handleSubmit} className="space-y-5">
          <div className="space-y-2">
            <label className="text-xs font-bold text-stone-300 block">
              رمز عبور فعلی:
            </label>
            <input
              type="password"
              value={oldPass}
              onChange={(e) => setOldPass(e.target.value)}
              className="w-full bg-stone-950 border border-stone-800 focus:border-gold-500 focus:ring-1 focus:ring-gold-500 rounded-xl px-4 py-3 text-sm text-stone-100 outline-none transition-all text-left font-sans"
              dir="ltr"
            />
          </div>

          <div className="space-y-2">
            <label className="text-xs font-bold text-stone-300 block">
              رمز عبور جدید:
            </label>
            <input
              type="password"

"use server";

import { cookies } from "next/headers";
import { redirect } from "next/navigation";

let adminPasswordStore = "admin123";

export async function loginAdminAction(formData: FormData) {
  const username = formData.get("username") as string;
  const password = formData.get("password") as string;

  if (username === "admin" && password === adminPasswordStore) {
    const cookieStore = await cookies();
    cookieStore.set("admin_session", "authenticated_token_xyz123", {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "strict",
      maxAge: 60 * 60 * 24,
      path: "/",
    });
    return { success: true };
  }

  return { success: false, error: "نام کاربری یا رمز عبور اشتباه است." };
}

export async function logoutAdminAction() {
  const cookieStore = await cookies();
  cookieStore.delete("admin_session");
  redirect("/");
}

  const cookieStore = await cookies();

}

export async function updateAdminPasswordAction(payload: {
  oldPass: string;
  newPass: string;
}) {
  try {
    const { oldPass, newPass } = payload;

    if (!oldPass || !newPass) {
      return { success: false, error: "پر کردن تمامی فیلدها الزامی است." };
    }

    if (oldPass !== adminPasswordStore) {
      return { success: false, error: "رمز عبور فعلی نادرست است." };
    }

    if (newPass.length < 6) {
      return {
        success: false,
        error: "رمز عبور جدید باید حداقل ۶ کاراکتر باشد.",
      };
    }

    adminPasswordStore = newPass;

    return {
      success: true,
      message: "رمز عبور شما با موفقیت تغییر یافت.",
    };
  } catch (error) {
    return { success: false, error: "خطا در تغییر رمز عبور رخ داد." };
  }
}

export async function submitOrderAction(payload: {
  productId: string;
  weightGrams: number;
  quantity: number;
  blend: string;
  grind: string;
  customerName: string;
  customerPhone: string;
  customerAddress: string;
  otp: string;
}): Promise<ActionResponse<Order>> {
  try {
    const {
      productId,
      weightGrams,
      quantity,
      blend,
      grind,
      customerName,
      customerPhone,
      customerAddress,
      otp,
    } = payload;

    if (
      !productId ||
      !weightGrams ||
      !quantity ||
      !blend ||
      !grind ||
      !customerName ||
      !customerPhone ||
      !customerAddress ||
      !otp
    ) {
      return {
        success: false,
        error: "اطلاعات ارسالی برای ثبت سفارش ناقص است.",
      };

        success: false,

      return {
        success: false,
        error: "محصول مورد نظر یافت نشد.",
      };
    }

    const blendProfile = getBlendProfile(blend);
    const priceMetrics = calculatePriceMetrics(
      product.pricing,
      blendProfile,
      Number(weightGrams),
      Number(quantity),
    );

    const secureFinalPrice = priceMetrics.finalPrice;

    const newOrder = createOrder({
      productId,
      weightGrams: Number(weightGrams),
      quantity: Number(quantity),
      grindType: `ترکیب: ${blend} | آسیاب: ${grind}`,
      finalPrice: secureFinalPrice,
      customerName,
      customerPhone,
      customerAddress,
    });

    getOrCreateUser(customerPhone, customerName);

    const cookieStore = await cookies();
    cookieStore.set("user_phone", customerPhone, {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "strict",
      maxAge: 60 * 60 * 24 * 30,
      path: "/",
    });

    revalidatePath("/");
    revalidatePath("/profile");

    return {
      success: true,
      data: newOrder,
    };
  } catch (error) {
    return {
      success: false,
      error: "خطایی در فرآیند محاسبات و ثبت نهایی سفارش رخ داد.",
    };
  }
}

      return {
        success: false,

      };

import React, { useState, useEffect, useTransition } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useCheckout } from "@/context/CheckoutContext";
import {
  checkoutSchema,
  CheckoutFormValues,
} from "@/components/checkout/schema";
import { submitOrderAction } from "@/app/actions/order";
import {
  getCurrentUserAction,
  checkPhoneRegisteredAction,
  loginUserWithOtpAction,
} from "@/app/actions/auth";

export type Step = "user_info" | "shipping" | "otp" | "success";

export function useCheckoutForm() {
  const { isOpen, checkoutData, closeCheckout } = useCheckout();
  const [step, setStep] = useState<Step>("user_info");
  const [isPending, startTransition] = useTransition();

  const [isLoginMode, setIsLoginMode] = useState(false);
  const [otpPurpose, setOtpPurpose] = useState<"login" | "checkout">(
    "checkout",
  );
  const [isLoggedInUser, setIsLoggedInUser] = useState(false);

  const [isSubmitting, setIsSubmitting] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [fakeTrackingCode, setFakeTrackingCode] = useState("");

  const productId = checkoutData?.productId || "";
  const selectedWeight = checkoutData?.selectedWeight || 0;
  const selectedBlend = checkoutData?.selectedBlend || "";
  const selectedGrind = checkoutData?.selectedGrind || "";
  const quantity = checkoutData?.quantity || 1;
  const finalPrice = checkoutData?.finalPrice || 0;

  const methods = useForm<CheckoutFormValues>({
    resolver: zodResolver(checkoutSchema),
    defaultValues: {
      firstName: "",
      lastName: "",
      phone: "",
      shippingAddress: "",
      otp: "",
    },
    mode: "onTouched",
  });

  const { trigger, getValues, reset, setValue } = methods;

  useEffect(() => {
    if (isOpen) {
      startTransition(async () => {
        const res = await getCurrentUserAction();
        if (res.success && res.user) {
          setIsLoggedInUser(true);
          const nameParts = res.user.name.split(" ");
          setValue("firstName", nameParts[0] || "");
          setValue("lastName", nameParts.slice(1).join(" ") || "");
          setValue("phone", res.user.phone);
          setValue("shippingAddress", res.user.address || "");
          setStep("shipping");
        } else {
          setIsLoggedInUser(false);
          setStep("user_info");
        }
      });
    }
  }, [isOpen, setValue]);

  const handleReset = () => {
    setStep("user_info");
    setIsLoginMode(false);
    setOtpPurpose("checkout");
    setIsLoggedInUser(false);
    setErrorMessage(null);
    reset();
  };

  const handleClose = () => {
    closeCheckout();
    setTimeout(handleReset, 300);
  };

  const handleUserInfoNext = async (e: React.FormEvent) => {
    e.preventDefault();
    setErrorMessage(null);

    if (isLoginMode) {
      const isPhoneValid = await trigger(["phone"]);
      if (!isPhoneValid) return;

      const phoneVal = getValues("phone");
      startTransition(async () => {
        const checkRes = await checkPhoneRegisteredAction({ phone: phoneVal });
        if (checkRes.success && checkRes.registered && checkRes.user) {
          const nameParts = checkRes.user.name.split(" ");
          setValue("firstName", nameParts[0] || "");
          setValue("lastName", nameParts.slice(1).join(" ") || "");
          setValue("shippingAddress", checkRes.user.address || "");
          setOtpPurpose("login");
          setStep("otp");
        } else {
          setErrorMessage(
            "این شماره همراه قبلاً ثبت‌نام نکرده است. لطفاً ثبت‌نام کنید.",
          );
        }
      });
    } else {
      const isFieldsValid = await trigger(["firstName", "lastName", "phone"]);
      if (isFieldsValid) {
        setOtpPurpose("checkout");
        setStep("shipping");
      }
    }
  };

  const handleShippingNext = async (e: React.FormEvent) => {
    e.preventDefault();
    const isAddressValid = await trigger(["shippingAddress"]);
    if (!isAddressValid) return;

    setErrorMessage(null);

    if (isLoggedInUser) {
      setIsSubmitting(true);
      const values = getValues();
      try {
        const result = await submitOrderAction({
          productId,
          weightGrams: selectedWeight,
          quantity,
          blend: selectedBlend,
          grind: selectedGrind,
          customerName: `${values.firstName} ${values.lastName}`,
          customerPhone: values.phone,
          customerAddress: values.shippingAddress,
          otp: "1234",
        });

        if (!result.success || !result.data) {
          throw new Error(result.error || "خطایی در ثبت سفارش رخ داد.");
        }

        setFakeTrackingCode(result.data.id);
        setStep("success");
      } catch (err) {
        setErrorMessage(
          err instanceof Error ? err.message : "خطا در برقراری ارتباط با سرور.",
        );
      } finally {
        setIsSubmitting(false);
      }
    } else {
      setStep("otp");
    }
  };

  const handleOtpVerifyAndSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    const isOtpValid = await trigger(["otp"]);
    if (!isOtpValid) return;

    setIsSubmitting(true);
    setErrorMessage(null);

    const values = getValues();

    try {
      if (otpPurpose === "login") {
        const loginRes = await loginUserWithOtpAction({
          phone: values.phone,
          otp: values.otp || "",
        });

        if (loginRes.success) {
          setIsLoggedInUser(true);
          setIsLoginMode(false);
          setValue("otp", "");
          setStep("shipping");
        } else {
          throw new Error(loginRes.error || "کد تایید اشتباه است.");
        }
      } else {
        const result = await submitOrderAction({
          productId,
          weightGrams: selectedWeight,
          quantity,
          blend: selectedBlend,
          grind: selectedGrind,
          customerName: `${values.firstName} ${values.lastName}`,
          customerPhone: values.phone,
          customerAddress: values.shippingAddress,
          otp: values.otp || "",
        });

        if (!result.success || !result.data) {
          throw new Error(result.error || "کد تایید اشتباه است.");
        }

        setFakeTrackingCode(result.data.id);
        setStep("success");
      }
    } catch (err) {
      setErrorMessage(
        err instanceof Error ? err.message : "خطا در پردازش اطلاعات.",
      );
    } finally {
      setIsSubmitting(false);
    }
  };

  return {
    isOpen,
    step,
    setStep,
    isPending,
    isLoginMode,
    setIsLoginMode,
    otpPurpose,
    isLoggedInUser,
    isSubmitting,
    errorMessage,
    setErrorMessage,
    fakeTrackingCode,
    selectedWeight,
    quantity,
    finalPrice,
    methods,
    handleClose,
    handleUserInfoNext,
    handleShippingNext,
    handleOtpVerifyAndSubmit,
  };
}

import { useState } from "react";
import { Comment } from "@/lib/mock-db";

import {
  submitCommentAction,
  deleteCommentAction,
  approveCommentAction,
  rejectCommentAction,
  editCommentAction,
} from "@/app/actions/comment";

export function useAdminComments(initialComments: Comment[]) {
  const [comments, setComments] = useState<Comment[]>(initialComments);
  const [replyTargetId, setReplyTargetId] = useState<string | null>(null);
  const [editTargetId, setEditTargetId] = useState<string | null>(null);
  const [editText, setEditText] = useState("");
  const [adminName, setAdminName] = useState("پشتیبان کافه راز");
  const [replyText, setReplyText] = useState("");
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [statusFilter, setStatusFilter] = useState<
    "all" | "pending" | "approved" | "rejected"
  >("all");

  const pendingCount = comments.filter(
    (c) => c.status === "pending" || !c.status,
  ).length;

  const filteredComments = comments.filter((comment) => {
    const status = comment.status || "pending";
    if (statusFilter === "all") return true;
    return status === statusFilter;
  });

  const handleDelete = async (commentId: string) => {
    try {
      const result = await deleteCommentAction(commentId);
      if (result.success) {
        setComments((prev) => {
          const filterNode = (list: Comment[]): Comment[] => {
            return list
              .filter((c) => c.id !== commentId)
              .map((c) => ({
                ...c,
                replies: c.replies ? filterNode(c.replies) : [],
              }));
          };
          return filterNode(prev);
        });
        toast.success("دیدگاه با موفقیت حذف شد.");
      } else {
        toast.error(result.error || "خطا در حذف دیدگاه.");
      }
    } catch {
      toast.error("خطای سیستمی رخ داد.");
    }
  };

  const handleStatusChange = async (
    commentId: string,
    newStatus: "approved" | "rejected",
  ) => {
    try {
      const result =
        newStatus === "approved"
          ? await approveCommentAction(commentId)
          : await rejectCommentAction(commentId);

      if (result.success) {
        toast.success(result.message || "وضعیت دیدگاه تغییر یافت.");
        setComments((prev) => {
          const updateStatus = (list: Comment[]): Comment[] => {
            return list.map((c) => {
              if (c.id === commentId) {
                return { ...c, status: newStatus };
              }
              return {
                ...c,
                replies: c.replies ? updateStatus(c.replies) : [],
              };
            });
          };
          return updateStatus(prev);
        });
      } else {
        toast.error(result.error || "خطا در تغییر وضعیت دیدگاه.");
      }
    } catch {
      toast.error("خطای ارتباطی با سرور رخ داد.");
    }
  };

  const handleEditSubmit = async (commentId: string) => {
    if (!editText.trim()) {
      toast.error("متن دیدگاه نمی‌تواند خالی باشد.");
      return;
    }

    try {
      const result = await editCommentAction(commentId, editText);
      if (result.success) {
        toast.success("تغییرات با موفقیت ذخیره شد.");
        setComments((prev) => {
          const updateText = (list: Comment[]): Comment[] => {
            return list.map((c) => {
              if (c.id === commentId) {
                return { ...c, comment_text: editText };
              }
              return {
                ...c,
                replies: c.replies ? updateText(c.replies) : [],
              };
            });
          };
          return updateText(prev);
        });
        setEditTargetId(null);
      } else {
        toast.error(result.error || "خطا در ویرایش دیدگاه.");
      }
    } catch {
      toast.error("خطا در برقراری ارتباط با سرور.");
    }
  };

  const handleReplySubmit = async (
    e: React.FormEvent,
    parentId: string,
    productId: string,
  ) => {
    e.preventDefault();
    if (!replyText.trim() || !adminName.trim()) {
      toast.error("لطفاً تمامی فیلدها را پر کنید.");
      return;
    }

    setIsSubmitting(true);
    try {
      const result = await submitCommentAction({
        productId,
        firstName: adminName,
        commentText: replyText,
        rating: null,
        parentId,
      });

      if (result.success && result.data) {
        const newReply = result.data;
        setComments((prev) =>
          prev.map((c) => {
            if (c.id === parentId) {
              return {
                ...c,
                replies: [...(c.replies || []), newReply],
              };
            }
            return c;
          }),
        );
        toast.success("پاسخ شما با موفقیت ثبت شد.");
        setReplyText("");
        setReplyTargetId(null);
      } else {
        toast.error("خطا در ثبت پاسخ.");
      }
    } catch {
      toast.error("خطای غیرمنتظره رخ داد.");
    } finally {
      setIsSubmitting(false);
    }
  };

  return {
    comments,
    filteredComments,
    pendingCount,
    statusFilter,
    setStatusFilter,
    replyTargetId,
    setReplyTargetId,
    editTargetId,
    setEditTargetId,
    editText,
    setEditText,
    adminName,
    setAdminName,
    replyText,
    setReplyText,
    isSubmitting,

export interface ProductPricing {
  pricePerKgArabica: number;
  pricePerKgRobusta: number;
}

export interface CoffeeProduct {
  id: string;
  name: string;
  description: string;
  stock_grams: number;
  pricing: ProductPricing | null;
}

export function useCoffeeProduct(initialProduct: CoffeeProduct) {
  const selection = useCoffeeSelection();

  const blendProfile = useMemo(() => {
    return getBlendProfile(selection.selectedBlend);
  }, [selection.selectedBlend]);

  const computedWeight = useMemo(() => {
    return selection.isCustomWeight
      ? selection.customWeightVal
      : selection.selectedWeight;
  }, [
    selection.isCustomWeight,
    selection.selectedWeight,
    selection.customWeightVal,
  ]);

  const priceMetrics = useMemo(() => {
    return calculatePriceMetrics(
      initialProduct.pricing,
      blendProfile,
      computedWeight,
      selection.quantity,
    );
  }, [
    initialProduct.pricing,
    blendProfile,
    computedWeight,
    selection.quantity,
  ]);

  const remainingStockCount = useMemo(() => {
    return calculateRemainingStock(initialProduct.stock_grams, computedWeight);
  }, [initialProduct.stock_grams, computedWeight]);

  const isOutOfStock = selection.quantity > remainingStockCount;

  const variantDetailsSnapshot = useMemo(() => {
    return `ترکیب: ${selection.selectedBlend} | آسیاب: ${selection.selectedGrind}`;
  }, [selection.selectedBlend, selection.selectedGrind]);

  return {
    ...selection,
    blendProfile,
    computedWeight,
    priceMetrics,
    remainingStockCount,
    isOutOfStock,
    variantDetailsSnapshot,
  };
}

  const [isCheckoutOpen, setIsCheckoutOpen] = useState<boolean>(false);

  return {
    selectedWeight,
    setSelectedWeight,
    isCustomWeight,
    setIsCustomWeight,
    customWeightVal,
    setCustomWeightVal,
    selectedBlend,
    setSelectedBlend,
    selectedGrind,
    setSelectedGrind,
    quantity,
    setQuantity,
    isCheckoutOpen,
    setIsCheckoutOpen,
  };
}
